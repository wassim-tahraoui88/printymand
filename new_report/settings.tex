    }{%
        \IfFileExists{figures/#2}{%
            \includegraphics[width=0.82\textwidth]{figures/#2}%
        }{%
            \IfFileExists{../figures/#2}{%
                \includegraphics[width=0.82\textwidth]{../figures/#2}%
            }{%
                \fbox{\parbox{0.82\textwidth}{\centering Missing figure file: #2}}%
            }%
        }%
    }%
    \caption{#3}
    \label{fig:#1}
\end{figure}
